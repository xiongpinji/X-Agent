## 第一部分：记忆系统架构回顾

### 1.1 多层记忆模型

X-Agent 采用分层记忆模型：

```
┌─────────────────────────────────────┐
│      即时记忆 (Immediate Memory)     │
│  - 当前会话上下文                    │
│  - 最近操作历史                      │
│  - TTL: 分钟级                       │
└─────────────────────────────────────┘
                  ↓
┌─────────────────────────────────────┐
│      短期记忆 (Short-term Memory)    │
│  - 会话级别记忆                      │
│  - 任务执行经验                      │
│  - TTL: 小时级                       │
└─────────────────────────────────────┘
                  ↓
┌─────────────────────────────────────┐
│      长期记忆 (Long-term Memory)     │
│  - 沉淀的知识和经验                  │
│  - 学习的模式和规则                  │
│  - TTL: 永久                         │
└─────────────────────────────────────┘
```

### 1.2 记忆存储后端

| 后端 | 特点 | 适用场景 |
|------|------|---------|
| PostgreSQL | 结构化、持久化 | 关键数据、审计日志 |
| Qdrant | 向量检索、高效 | 语义搜索、相似度匹配 |
| Neo4j | 图存储、关系 | 知识图谱、关系分析 |
| Redis | 内存、高速 | 缓存、会话数据 |

## 第三部分：高级记忆策略

### 3.1 混合检索策略

```python
class HybridMemoryRetriever:
    def __init__(self, config):
        self.config = config
        self.semantic_retriever = SemanticRetriever(config)
        self.keyword_retriever = KeywordRetriever(config)
        self.graph_retriever = GraphRetriever(config)
    
    async def retrieve(self, query, context=None):
        """混合检索"""
        # 并行执行多种检索
        semantic_results = await self.semantic_retriever.retrieve(query)
        keyword_results = await self.keyword_retriever.retrieve(query)
        graph_results = await self.graph_retriever.retrieve(query, context)
        
        # 融合结果
        fused_results = self._fuse_results(
            semantic_results,
            keyword_results,
            graph_results
        )
        
        return fused_results
    
    def _fuse_results(self, semantic, keyword, graph):
        """融合多种检索结果"""
        # 使用加权融合
        weights = {
            "semantic": 0.5,
            "keyword": 0.3,
            "graph": 0.2
        }
        
        fused = {}
        for result in semantic + keyword + graph:
            item_id = result["id"]
            if item_id not in fused:
                fused[item_id] = {
                    "item": result,
                    "score": 0
                }
            
            # 根据来源加权
            source = result.get("source", "unknown")
            fused[item_id]["score"] += weights.get(source, 0) * result.get("score", 0)
        
        # 按分数排序
        sorted_results = sorted(
            fused.values(),
            key=lambda x: x["score"],
            reverse=True
        )
        
        return [r["item"] for r in sorted_results]
```

### 3.2 记忆污染检测

```python
class MemoryPollutionDetector:
    def __init__(self, config):
        self.config = config
        self.anomaly_detector = AnomalyDetector()
        self.conflict_detector = ConflictDetector()
    
    async def detect_pollution(self, memory_items):
        """检测记忆污染"""
        pollution_report = {
            "anomalies": [],
            "conflicts": [],
            "inconsistencies": [],
            "pollution_score": 0
        }
        
        # 检测异常
        anomalies = await self.anomaly_detector.detect(memory_items)
        pollution_report["anomalies"] = anomalies
        
        # 检测冲突
        conflicts = await self.conflict_detector.detect(memory_items)
        pollution_report["conflicts"] = conflicts
        
        # 检测不一致
        inconsistencies = self._detect_inconsistencies(memory_items)
        pollution_report["inconsistencies"] = inconsistencies
        
        # 计算污染分数
        pollution_report["pollution_score"] = self._calculate_pollution_score(
            anomalies,
            conflicts,
            inconsistencies
        )
        
        return pollution_report
    
    def _detect_inconsistencies(self, memory_items):
        """检测不一致"""
        inconsistencies = []
        
        for i, item1 in enumerate(memory_items):
            for item2 in memory_items[i+1:]:
                if self._is_inconsistent(item1, item2):
                    inconsistencies.append({
                        "item1_id": item1["id"],
                        "item2_id": item2["id"],
                        "reason": "conflicting_information"
                    })
        
        return inconsistencies
```

class MemoryVersionManager:
    def __init__(self, config):
        self.config = config
        self.storage = MemoryStorage(config)
    
    async def create_version(self, memory_item, version_info):
        """创建记忆版本"""
        version = {
            "id": generate_id(),
            "memory_id": memory_item["id"],
            "version_number": await self._get_next_version_number(memory_item["id"]),
            "content": memory_item,
            "created_at": datetime.now(),

            "reason": version_info.get("reason"),
            "metadata": version_info.get("metadata", {})
        }
        
        await self.storage.save_version(version)
        return version
    
    async def rollback_to_version(self, memory_id, version_number):
        """回滚到指定版本"""
        version = await self.storage.get_version(memory_id, version_number)
        
        if not version:
            raise VersionNotFoundError(f"Version {version_number} not found")
        
        # 创建回滚记录
        rollback_record = {
            "memory_id": memory_id,
            "from_version": await self._get_current_version_number(memory_id),
            "to_version": version_number,
            "timestamp": datetime.now()
        }
        
        # 恢复内容
        await self.storage.update_memory(memory_id, version["content"])
        await self.storage.save_rollback_record(rollback_record)
        
        return version
    
    async def get_version_history(self, memory_id):
        """获取版本历史"""
        return await self.storage.get_versions(memory_id)

## 第四部分：性能优化

### 4.1 缓存策略

```python
class MemoryCacheStrategy:
    def __init__(self, config):
        self.config = config
        self.cache = {}
        self.cache_stats = {
            "hits": 0,
            "misses": 0,
            "evictions": 0
        }
    
    async def get_with_cache(self, key, fetch_func):
        """带缓存的获取"""
        # 检查缓存
        if key in self.cache:
            self.cache_stats["hits"] += 1
            return self.cache[key]
        
        # 缓存未命中
        self.cache_stats["misses"] += 1
        value = await fetch_func()
        
        # 存入缓存
        self._put_in_cache(key, value)
        
        return value
    
    def _put_in_cache(self, key, value):
        """存入缓存"""
        # 检查缓存大小
        if len(self.cache) >= self.config.get("max_cache_size", 10000):
            # 执行缓存驱逐
            self._evict_cache()
        
        self.cache[key] = {
            "value": value,
            "timestamp": time.time(),
            "access_count": 0
        }
    
    def _evict_cache(self):
        """驱逐缓存"""
        # 使用 LRU 策略
        lru_key = min(
            self.cache.keys(),
            key=lambda k: self.cache[k]["timestamp"]
        )
        del self.cache[lru_key]
        self.cache_stats["evictions"] += 1
```

### 4.2 批量操作优化

```python
class MemoryBatchOptimizer:
    def __init__(self, config):
        self.config = config
        self.batch_size = config.get("batch_size", 100)
    
    async def batch_write(self, memory_items):
        """批量写入"""
        for i in range(0, len(memory_items), self.batch_size):
            batch = memory_items[i:i+self.batch_size]
            await self._write_batch(batch)
    
    async def batch_retrieve(self, queries):
        """批量检索"""
        results = []
        for i in range(0, len(queries), self.batch_size):
            batch = queries[i:i+self.batch_size]
            batch_results = await self._retrieve_batch(batch)
            results.extend(batch_results)
        return results
    
    async def _write_batch(self, batch):
        """写入一个批次"""
        # 使用数据库批量插入
        await self.storage.batch_insert(batch)
    
    async def _retrieve_batch(self, batch):
        """检索一个批次"""
        # 使用数据库批量查询
        return await self.storage.batch_query(batch)
```

### 5.1 访问控制

```python
class MemoryAccessControl:
    def __init__(self, config):
        self.config = config
    
    async def check_access(self, user, memory_item, action):
        """检查访问权限"""
        # 检查用户权限
        if not await self._has_permission(user, action):
            raise PermissionDenied(f"User {user.id} lacks {action} permission")
        
        # 检查作用域
        if not await self._check_scope(user, memory_item):
            raise ScopeViolation(f"Memory item {memory_item.id} is out of scope")
        
        # 检查租户隔离
        if not await self._check_tenant_isolation(user, memory_item):
            raise TenantViolation(f"Tenant mismatch for memory item {memory_item.id}")
        
        return True
    
    async def _has_permission(self, user, action):
        """检查用户权限"""
        permissions = await self._get_user_permissions(user)
        return action in permissions
    
    async def _check_scope(self, user, memory_item):
        """检查作用域"""
        user_scope = user.get_scope()
        item_scope = memory_item.get_scope()
        return self._is_scope_compatible(user_scope, item_scope)
    
    async def _check_tenant_isolation(self, user, memory_item):
        """检查租户隔离"""
        return user.tenant_id == memory_item.tenant_id
```

### 5.3 数据保留策略

```python
class MemoryRetentionPolicy:
    def __init__(self, config):
        self.config = config
    
    async def apply_retention_policy(self):
        """应用数据保留策略"""
        policies = self.config.get("retention_policies", [])
        
        for policy in policies:
            if policy["type"] == "age_based":
                await self._apply_age_based_policy(policy)
            elif policy["type"] == "size_based":
                await self._apply_size_based_policy(policy)
            elif policy["type"] == "importance_based":
                await self._apply_importance_based_policy(policy)
    
    async def _apply_age_based_policy(self, policy):
        """应用基于年龄的保留策略"""
        max_age = policy.get("max_age")
        cutoff_date = datetime.now() - timedelta(seconds=max_age)
        
        # 删除超过最大年龄的记忆
        await self.storage.delete_older_than(cutoff_date)
    
    async def _apply_size_based_policy(self, policy):
        """应用基于大小的保留策略"""
        max_size = policy.get("max_size")
        current_size = await self.storage.get_total_size()
        
        if current_size > max_size:
            # 删除最旧的记忆直到达到目标大小
            excess = current_size - max_size
            await self.storage.delete_oldest_until_size(excess)
    
    async def _apply_importance_based_policy(self, policy):
        """应用基于重要性的保留策略"""
        min_importance = policy.get("min_importance")
        
        # 删除重要性低于阈值的记忆
        await self.storage.delete_below_importance(min_importance)
```

## 第六部分：最佳实践

### 6.1 记忆设计原则

1. **粒度适当**：记忆项不应过大或过小
2. **结构清晰**：使用明确的数据结构
3. **可检索性**：确保记忆可以被有效检索
4. **可维护性**：记忆应该易于更新和维护
5. **隐私保护**：敏感信息应该加密存储

### 6.2 常见陷阱

| 陷阱 | 问题 | 解决方案 |
|------|------|---------|
| 记忆爆炸 | 记忆数量过多 | 实现记忆巩固和清理 |
| 检索缓慢 | 检索性能差 | 优化索引和缓存 |
| 记忆污染 | 记忆数据不准确 | 实现污染检测和清理 |
| 隐私泄露 | 敏感信息泄露 | 实现访问控制和加密 |
| 存储溢出 | 存储空间不足 | 实现数据保留策略 |

### 6.3 性能基准

| 指标 | 目标 | 说明 |
|------|------|------|
| 写入延迟 | < 100ms | 单个记忆项写入时间 |
| 检索延迟 | < 500ms | 单次检索时间 |
| 批量写入吞吐 | > 10k/s | 每秒写入记忆项数 |
| 检索准确率 | > 0.9 | 检索结果相关性 |
| 存储效率 | > 0.8 | 实际使用空间与总空间比率 |
