```python
class DynamicContentHandler:
    def __init__(self, browser_session):
        self.browser = browser_session
    
    async def wait_for_dynamic_content(self, selector, timeout=10):
        """等待动态内容加载"""
        start_time = time.time()
        
        while time.time() - start_time < timeout:
            try:
                # 尝试获取元素
                element = await self.browser.query_selector(selector)
                
                if element:
                    # 检查元素是否可见
                    is_visible = await self.browser.evaluate(
                        f"document.querySelector('{selector}').offsetParent !== null"
                    )
                    
                    if is_visible:
                        return element
                
                # 等待后重试
                await asyncio.sleep(0.5)
            except Exception as e:
                logger.debug(f"Waiting for element: {e}")
                await asyncio.sleep(0.5)
        
        raise TimeoutError(f"Element {selector} not found within {timeout}s")
    
    async def wait_for_network_idle(self, timeout=10):
        """等待网络空闲"""
        await self.browser.wait_for_load_state("networkidle", timeout=timeout*1000)
    
    async def wait_for_navigation(self, timeout=10):
        """等待页面导航完成"""
        await self.browser.wait_for_load_state("domcontentloaded", timeout=timeout*1000)
```

```python
class FormHandler:
    def __init__(self, browser_session):
        self.browser = browser_session
    
    async def fill_form(self, form_data):
        """填充表单"""
        for field_name, field_value in form_data.items():
            selector = f"[name='{field_name}']"
            
            # 获取元素类型
            element_type = await self.browser.evaluate(
                f"document.querySelector('{selector}').type"
            )
            
            if element_type == "text":
                await self.browser.fill(selector, field_value)
            elif element_type == "checkbox":
                if field_value:
                    await self.browser.check(selector)
                else:
                    await self.browser.uncheck(selector)
            elif element_type == "radio":
                await self.browser.check(selector)
            elif element_type == "select-one":
                await self.browser.select_option(selector, field_value)
            elif element_type == "textarea":
                await self.browser.fill(selector, field_value)
    
    async def submit_form(self, form_selector, wait_for_navigation=True):
        """提交表单"""
        if wait_for_navigation:
            async with self.browser.expect_navigation():
                await self.browser.click(f"{form_selector} button[type='submit']")
        else:
            await self.browser.click(f"{form_selector} button[type='submit']")
    
    async def extract_form_data(self, form_selector):
        """提取表单数据"""
        form_data = await self.browser.evaluate(
            f"""
            (() => {{
                const form = document.querySelector('{form_selector}');
                const formData = new FormData(form);
                const data = {{}};
                for (let [key, value] of formData.entries()) {{
                    data[key] = value;
                }}
                return data;
            }})()
            """
        )
        return form_data
```

#### 1. 并行浏览器操作

```python
class ParallelBrowserOperations:
    def __init__(self, max_concurrent=5):
        self.max_concurrent = max_concurrent
        self.semaphore = asyncio.Semaphore(max_concurrent)
    
    async def parallel_navigate(self, urls):
        """并行导航到多个 URL"""
        tasks = [self._navigate_with_limit(url) for url in urls]
        results = await asyncio.gather(*tasks)
        return results
    
    async def _navigate_with_limit(self, url):
        """带并发限制的导航"""
        async with self.semaphore:
            try:
                page = await self.browser.new_page()
                await page.goto(url)
                content = await page.content()
                await page.close()
                return {"url": url, "content": content, "status": "success"}
            except Exception as e:
                return {"url": url, "error": str(e), "status": "failed"}
    
    async def parallel_extract_data(self, pages, selectors):
        """并行提取数据"""
        tasks = [
            self._extract_with_limit(page, selectors)
            for page in pages
        ]
        results = await asyncio.gather(*tasks)
        return results
    
    async def _extract_with_limit(self, page, selectors):
        """带并发限制的数据提取"""
        async with self.semaphore:
            data = {}
            for selector_name, selector in selectors.items():
                try:
                    element = await page.query_selector(selector)
                    if element:
                        data[selector_name] = await element.text_content()
                except Exception as e:
                    data[selector_name] = None
            return data
```

```python
class BrowserCacheManager:
    def __init__(self, cache_dir="./browser_cache"):
        self.cache_dir = cache_dir
        self.page_cache = {}
    
    async def get_or_create_page(self, url, cache_key=None):
        """获取或创建页面（带缓存）"""
        if cache_key is None:
            cache_key = url
        
        # 检查缓存
        if cache_key in self.page_cache:
            cached_page = self.page_cache[cache_key]
            if await self._is_page_valid(cached_page):
                return cached_page
        
        # 创建新页面
        page = await self.browser.new_page()
        await page.goto(url)
        
        # 存入缓存
        self.page_cache[cache_key] = page
        
        return page
    
    async def _is_page_valid(self, page):
        """检查页面是否有效"""
        try:
            await page.evaluate("1")
            return True
        except:
            return False
    
    async def clear_cache(self):
        """清空缓存"""
        for page in self.page_cache.values():
            await page.close()
        self.page_cache.clear()
```

#### 1. 重试机制

```python
class BrowserRetryHandler:
    def __init__(self, max_retries=3, backoff_factor=2.0):
        self.max_retries = max_retries
        self.backoff_factor = backoff_factor
    
    async def retry_operation(self, operation, *args, **kwargs):
        """带重试的操作"""
        last_error = None
        
        for attempt in range(self.max_retries):
            try:
                return await operation(*args, **kwargs)
            except Exception as e:
                last_error = e
                
                if attempt < self.max_retries - 1:
                    wait_time = self.backoff_factor ** attempt
                    logger.warning(
                        f"Operation failed (attempt {attempt+1}/{self.max_retries}), "
                        f"retrying in {wait_time}s: {e}"
                    )
                    await asyncio.sleep(wait_time)
        
        raise last_error
    
    async def click_with_retry(self, page, selector):
        """带重试的点击"""
        async def click_operation():
            await page.click(selector)
        
        return await self.retry_operation(click_operation)
    
    async def fill_with_retry(self, page, selector, text):
        """带重试的填充"""
        async def fill_operation():
            await page.fill(selector, text)
        
        return await self.retry_operation(fill_operation)
```

```python
class BrowserErrorRecovery:
    def __init__(self, browser_session):
        self.browser = browser_session
    
    async def recover_from_crash(self):
        """从浏览器崩溃恢复"""
        try:
            # 尝试重新连接
            await self.browser.reconnect()
        except:
            # 重新启动浏览器
            await self.browser.restart()
    
    async def handle_navigation_error(self, url):
        """处理导航错误"""
        try:
            await self.browser.goto(url)
        except Exception as e:
            if "net::ERR_CONNECTION_REFUSED" in str(e):
                logger.error(f"Connection refused for {url}")
                # 重试或使用备用 URL
                await asyncio.sleep(5)
                await self.browser.goto(url)
            elif "net::ERR_NAME_NOT_RESOLVED" in str(e):
                logger.error(f"DNS resolution failed for {url}")
                raise
            else:
                raise
```

```python
class WindowManager:
    def __init__(self, desktop_session):
        self.desktop = desktop_session
    
    async def find_window(self, window_title=None, window_class=None):
        """查找窗口"""
        windows = await self.desktop.get_windows()
        
        for window in windows:
            if window_title and window_title in window.title:
                return window
            if window_class and window_class in window.class_name:
                return window
        
        return None
    
    async def activate_window(self, window):
```

```python
class ImageRecognition:
    def __init__(self, desktop_session):
        self.desktop = desktop_session
    
    async def find_image_on_screen(self, image_path, confidence=0.8):
        """在屏幕上查找图像"""
        screenshot = await self.desktop.take_screenshot()
        
        # 使用 OpenCV 进行图像匹配
        import cv2
        import numpy as np
        
        template = cv2.imread(image_path)
        result = cv2.matchTemplate(screenshot, template, cv2.TM_CCOEFF_NORMED)
        
        min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(result)
        
        if max_val >= confidence:
            return max_loc
        
        return None
    
    async def click_on_image(self, image_path, confidence=0.8):
        """点击图像位置"""
        location = await self.find_image_on_screen(image_path, confidence)
        
        if location:
            x, y = location
            # 点击图像中心
            template = cv2.imread(image_path)
            h, w = template.shape[:2]
            await self.desktop.click(x + w//2, y + h//2)
            return True
        
        return False
```

#### 1. 操作批处理

```python
class DesktopOperationBatcher:
    def __init__(self, desktop_session):
        self.desktop = desktop_session
        self.operations = []
    
    def add_operation(self, operation_type, *args, **kwargs):
        """添加操作到批处理队列"""
        self.operations.append({
            "type": operation_type,
            "args": args,
            "kwargs": kwargs
        })
    
    async def execute_batch(self):
        """执行批处理操作"""
        results = []
        
        for operation in self.operations:
            op_type = operation["type"]
            args = operation["args"]
            kwargs = operation["kwargs"]
            
            if op_type == "click":
                result = await self.desktop.click(*args, **kwargs)
            elif op_type == "type":
                result = await self.desktop.type_text(*args, **kwargs)
            elif op_type == "key_press":
                result = await self.desktop.key_press(*args, **kwargs)
            elif op_type == "move_mouse":
                result = await self.desktop.move_mouse(*args, **kwargs)
            
            results.append(result)
        
        self.operations.clear()
        return results
```

```python
class DesktopStateManager:
    def __init__(self, desktop_session):
        self.desktop = desktop_session
        self.state_cache = {}
    
    async def get_element_state(self, element_id, force_refresh=False):
        """获取元素状态（带缓存）"""
        if not force_refresh and element_id in self.state_cache:
            return self.state_cache[element_id]
        
        state = await self.desktop.get_element_state(element_id)
        self.state_cache[element_id] = state
        
        return state
    
    async def wait_for_state_change(self, element_id, timeout=10):
        """等待元素状态变化"""
        initial_state = await self.get_element_state(element_id)
        start_time = time.time()
        
        while time.time() - start_time < timeout:
            current_state = await self.get_element_state(element_id, force_refresh=True)
            
            if current_state != initial_state:
                return current_state
            
            await asyncio.sleep(0.5)
        
        raise TimeoutError(f"Element state did not change within {timeout}s")
```

```python
class DesktopErrorRecovery:
    def __init__(self, desktop_session):
        self.desktop = desktop_session
    
    async def handle_element_not_found(self, element_id, retry_count=3):
        """处理元素未找到错误"""
        for attempt in range(retry_count):
            try:
                element = await self.desktop.find_element(element_id)
                return element
            except ElementNotFoundError:
                if attempt < retry_count - 1:
                    # 刷新屏幕或重新扫描
                    await self.desktop.refresh_ui()
                    await asyncio.sleep(1)
                else:
                    raise
    
    async def handle_interaction_failure(self, operation, *args, **kwargs):
        """处理交互失败"""
        try:
            return await operation(*args, **kwargs)
        except InteractionFailedError as e:
            logger.warning(f"Interaction failed: {e}, attempting recovery")
            
            # 尝试恢复
            await self.desktop.reset_state()
            await asyncio.sleep(1)
            
            # 重试操作
            return await operation(*args, **kwargs)
```

## 第四部分：最佳实践

### 4.1 设计原则

1. **显式等待**：使用显式等待而不是固定延迟
2. **错误处理**：实现完整的错误处理和恢复
3. **资源管理**：正确管理浏览器和桌面会话
4. **性能优化**：使用并行和缓存提升性能
5. **可维护性**：编写清晰、可维护的代码

### 4.2 常见陷阱

| 陷阱 | 问题 | 解决方案 |
|------|------|---------|
| 固定延迟 | 不可靠的等待 | 使用显式等待 |
| 资源泄漏 | 未关闭会话 | 使用上下文管理器 |
| 脆弱的选择器 | 选择器易变 | 使用稳定的选择器 |
| 缺少错误处理 | 脚本崩溃 | 添加完整的错误处理 |
| 性能问题 | 执行缓慢 | 使用并行和缓存 |

### 4.3 性能基准

| 指标 | 目标 | 说明 |
|------|------|------|
| 页面加载时间 | < 5s | 平均页面加载时间 |
| 元素查找时间 | < 1s | 平均元素查找时间 |
| 点击响应时间 | < 500ms | 点击后的响应时间 |
| 并行操作吞吐 | > 10 ops/s | 每秒并行操作数 |
