### 2.3 代码质量标准

#### 1. 类型注解

```python
# 好的做法
async def process_data(data: List[Dict[str, Any]]) -> Dict[str, Any]:
    """处理数据"""
    pass

# 避免
async def process_data(data):
    """处理数据"""
    pass
```

#### 2. 错误处理

```python
# 好的做法
try:
    result = await operation()
except SpecificError as e:
    logger.error(f"Operation failed: {e}")
    raise
except Exception as e:
    logger.error(f"Unexpected error: {e}")
    raise

# 避免
try:
    result = await operation()
except:
    pass
```

#### 3. 日志记录

```python
# 好的做法
logger.info("Starting operation")
logger.debug(f"Operation parameters: {params}")
logger.error(f"Operation failed: {error}")

# 避免
print("Starting operation")
print(f"Operation parameters: {params}")
```

#### 4. 文档字符串

```python
# 好的做法
async def process_data(data: List[Dict]) -> Dict:
    """
    处理数据。
    
    Args:
        data: 输入数据列表
    
    Returns:
        处理后的数据字典
    
    Raises:
        ValueError: 如果数据格式无效
    
    Example:
        >>> result = await process_data([{"key": "value"}])
        >>> print(result)
        {"processed": True}
    """
    pass

# 避免
async def process_data(data):
    # 处理数据
    pass
```

## 第四部分：版本同步机制

### 4.1 API 变更追踪

```python
# examples/api_version_tracker.py

class APIVersionTracker:
    """追踪 API 版本变更"""
    
    def __init__(self):
        self.api_versions = {
            "agent": "1.0.0",
            "workflow": "1.0.0",
            "memory": "1.0.0"
        }
    
    def check_compatibility(self, example_file, required_versions):
        """检查示例与 API 版本的兼容性"""
        for api, required_version in required_versions.items():
            current_version = self.api_versions.get(api)
            
            if not self._is_compatible(current_version, required_version):
                raise VersionMismatchError(
                    f"Example requires {api} {required_version}, "
                    f"but current version is {current_version}"
                )
    
    def _is_compatible(self, current, required):
        """检查版本兼容性"""
        # 简单的版本比较逻辑
        current_parts = current.split(".")
        required_parts = required.split(".")
        
        # 主版本号必须相同
        return current_parts[0] == required_parts[0]
```

### 4.3 自动同步工具

```python
# tools/sync_examples.py

import yaml
import subprocess
from pathlib import Path


class ExampleSynchronizer:
    """同步示例代码与 API"""
    
    def __init__(self, examples_dir, metadata_file):
        self.examples_dir = Path(examples_dir)
        self.metadata_file = metadata_file
        self.metadata = self._load_metadata()
    
    def check_all_examples(self):
        """检查所有示例"""
        issues = []
        
        for example_meta in self.metadata["examples"]:
            example_file = self.examples_dir / f"{example_meta['id']}.py"
            
            # 检查文件是否存在
            if not example_file.exists():
                issues.append(f"Example file not found: {example_file}")
                continue
            
            # 检查 API 版本兼容性
            try:
                self._check_api_compatibility(example_file, example_meta)
            except VersionMismatchError as e:
                issues.append(str(e))
            
            # 检查依赖
            try:
                self._check_dependencies(example_meta)
            except DependencyError as e:
                issues.append(str(e))
        
        return issues
    
    def update_example(self, example_id, changes):
        """更新示例"""
        example_meta = self._find_example_meta(example_id)
        
        if not example_meta:
            raise ValueError(f"Example {example_id} not found")
        
        # 更新元数据
        example_meta.update(changes)
        example_meta["last_updated"] = datetime.now().isoformat()
        
        # 保存元数据
        self._save_metadata()
    
    def _load_metadata(self):
        """加载元数据"""
        with open(self.metadata_file, 'r') as f:
            return yaml.safe_load(f)
    
    def _save_metadata(self):
        """保存元数据"""
        with open(self.metadata_file, 'w') as f:
            yaml.dump(self.metadata, f)
```
